inner_loop(NumActions, _, []) :-
	max_agent_actions (N),

	!.

inner_loop(NumActions, Percept, [Action | Actions]) :-
	run_agent (Percept, Action),
	execute(Action, NewPercept),
	NumActions1 is NumActions + 1,
	inner_loop(NumActions1, NewPercept, Actions).
